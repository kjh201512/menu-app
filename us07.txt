package com.menuapp.recommendation;
import com.menuapp.profile.IngredientNormalizer;
import java.util.Collection;
import org.springframework.stereotype.Component;
@Component
public class FilterPolicy {
    public boolean allowed(Menu menu,Collection<String> restrictedIngredients) {
        // 식재료 정보가 없거나 불완전한 메뉴는 안전측으로 제외합니다.
        var names=menu.ingredientNames();
        return menu.isIngredientsComplete() && !names.isEmpty() && names.stream().noneMatch(
            i->restrictedIngredients.stream().anyMatch(r->IngredientNormalizer.conflicts(i,r)));
    }
}

// ===== backend/src/main/java/com/menuapp/recommendation/GeoDistance.java =====
package com.menuapp.recommendation;
public final class GeoDistance {
    public static final double EARTH_RADIUS_METERS=6371000.0;
    private GeoDistance() {}
    public static double meters(double lat1,double lng1,double lat2,double lng2) {
        double dLat=Math.toRadians(lat2-lat1),dLng=Math.toRadians(lng2-lng1);
        double a=Math.pow(Math.sin(dLat/2),2)+Math.cos(Math.toRadians(lat1))*Math.cos(Math.toRadians(lat2))*Math.pow(Math.sin(dLng/2),2);
        return 2*EARTH_RADIUS_METERS*Math.asin(Math.sqrt(Math.min(1,Math.max(0,a))));
    }
}

import java.util.*;
@Entity @Table(name="menu")
public class Menu {
    @Id @GeneratedValue(strategy=GenerationType.IDENTITY) @Column(name="menu_id") private Long id;
    @ManyToOne(fetch=FetchType.LAZY,optional=false) @JoinColumn(name="restaurant_id",nullable=false) private Restaurant restaurant;
    @Column(nullable=false,length=100) private String name;
    @Enumerated(EnumType.STRING) @Column(nullable=false,length=30) private Category category;
    @Column(nullable=false) private int price;
    @Column(name="is_signature",nullable=false) private boolean signature;
    @Column(name="ingredients_complete",nullable=false) private boolean ingredientsComplete;
    @OneToMany(mappedBy="menu",cascade=CascadeType.ALL,orphanRemoval=true) private List<MenuIngredient> ingredients=new ArrayList<>();
    protected Menu() {}
    public Menu(Restaurant restaurant,String name,Category category,int price,boolean complete,List<String> ingredients) {
        this.restaurant=restaurant;this.name=name;this.category=category;this.price=price;this.signature=true;this.ingredientsComplete=complete;
        ingredients.stream().map(IngredientNormalizer::normalize).distinct().forEach(i->this.ingredients.add(new MenuIngredient(this,i)));
    }
    public Long getId() { return id; }
    public Restaurant getRestaurant() { return restaurant; }
    public String getName() { return name; }
    public Category getCategory() { return category; }
    public int getPrice() { return price; }
    public boolean isIngredientsComplete() { return ingredientsComplete; }
    public List<String> ingredientNames() { return ingredients.stream().map(MenuIngredient::getIngredientName).sorted().toList(); }
}

package com.menuapp.recommendation;
import com.menuapp.common.ApiException;
import com.menuapp.profile.*;
import com.menuapp.user.*;
import java.util.*;
import java.util.stream.Collectors;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
@Service
public class RecommendationService {
    public record Card(Long menuId,Long restaurantId,String menuName,String restaurantName,Category category,
                       int price,long distanceMeters,boolean preferred,List<String> ingredients) {}
    public record Result(String sessionId,int count,int availableCount,String message,List<Card> items) {}
    private record Candidate(Menu menu,double distance,double weight) {}
    private final UserRepository users;
    private final UserPreferenceRepository preferences;
    private final UserRestrictionRepository restrictions;
    private final MenuRepository menus;
    private final RecommendationLogRepository logs;
    private final FilterPolicy filter;
    public RecommendationService(UserRepository users,UserPreferenceRepository preferences,UserRestrictionRepository restrictions,
                                 MenuRepository menus,RecommendationLogRepository logs,FilterPolicy filter) {
        this.users=users;this.preferences=preferences;this.restrictions=restrictions;this.menus=menus;this.logs=logs;this.filter=filter;
    }
    @Transactional
    public Result recommend(Long userId,double lat,double lng) {
        if (!Double.isFinite(lat) || !Double.isFinite(lng) || lat < -90 || lat > 90 || lng < -180 || lng > 180)
            throw new ApiException(HttpStatus.BAD_REQUEST,"INVALID_COORDINATES","위도는 -90~90, 경도는 -180~180의 유한한 숫자여야 합니다.");
        User user=users.findById(userId).orElseThrow(()->new ApiException(HttpStatus.UNAUTHORIZED,"UNAUTHORIZED","다시 로그인해 주세요."));
        var weights=preferences.findForUser(userId).stream().collect(Collectors.toMap(UserPreference::getCategory,UserPreference::getWeight));
        var blocked=restrictions.findForUser(userId).stream().map(UserRestriction::getIngredientName).toList();
        double latDelta=Math.toDegrees(1000.0/GeoDistance.EARTH_RADIUS_METERS);
        var candidates=menus.findCandidates(Math.max(-90,lat-latDelta),Math.min(90,lat+latDelta)).stream()
            .filter(m->filter.allowed(m,blocked))
            .map(m->new Candidate(m,GeoDistance.meters(lat,lng,m.getRestaurant().getLatitude(),m.getRestaurant().getLongitude()),weights.getOrDefault(m.getCategory(),0.0)))
            .filter(c->c.distance()<=1000.0)
            .sorted(Comparator.comparingDouble(Candidate::weight).reversed().thenComparingDouble(Candidate::distance).thenComparing(c->c.menu().getId()))
            .toList();
        String sessionId=UUID.randomUUID().toString();
        var selected=candidates.stream().limit(3).toList();
        logs.saveAll(selected.stream().map(c->new RecommendationLog(user,c.menu(),sessionId)).toList());
        var cards=selected.stream().map(c->new Card(c.menu().getId(),c.menu().getRestaurant().getId(),c.menu().getName(),
            c.menu().getRestaurant().getName(),c.menu().getCategory(),c.menu().getPrice(),Math.round(c.distance()),
            weights.containsKey(c.menu().getCategory()),c.menu().ingredientNames())).toList();
        String message=cards.isEmpty()?"조건에 맞는 메뉴가 없습니다. 알레르기 필터는 완화하지 않았습니다.":
            cards.size()<3?"조건에 맞는 메뉴가 "+cards.size()+"개뿐입니다. 가능한 메뉴만 표시합니다.":"등록된 제한 식재료를 제외한 메뉴입니다.";
        return new Result(sessionId,cards.size(),candidates.size(),message,cards);
    }
}

package com.menuapp.profile;
import java.text.Normalizer;
import java.util.Locale;
import java.util.Map;
/** 데이터에 명시된 식재료의 보수적 정규화일 뿐 의료용 알레르겐 판별기가 아닙니다. */
public final class IngredientNormalizer {
    private IngredientNormalizer() {}
    private static final Map<String, String> ALIASES = Map.ofEntries(
        Map.entry("계란", "달걀"), Map.entry("난류", "달걀"), Map.entry("egg", "달걀"), Map.entry("eggs", "달걀"),
        Map.entry("milk", "우유"), Map.entry("밀가루", "밀"), Map.entry("wheat", "밀"),
        Map.entry("콩", "대두"), Map.entry("soy", "대두"), Map.entry("soybean", "대두"),
        Map.entry("peanut", "땅콩"), Map.entry("shrimp", "새우"), Map.entry("pork", "돼지고기"), Map.entry("beef", "소고기")
    );
    public static String normalize(String raw) {
        if (raw == null) return "";
        StringBuilder clean = new StringBuilder();
        Normalizer.normalize(raw, Normalizer.Form.NFKC).toLowerCase(Locale.ROOT).codePoints()
            .filter(c -> !Character.isWhitespace(c) && !Character.isSpaceChar(c)).forEach(clean::appendCodePoint);
        String value = clean.toString();
        return ALIASES.getOrDefault(value, value);
    }
    public static boolean conflicts(String ingredient, String restriction) {
        String a = normalize(ingredient), b = normalize(restriction);
        return !a.isEmpty() && !b.isEmpty() && (a.contains(b) || b.contains(a));
    }
}

export class ApiError extends Error {
  constructor(public status: number, public code: string, message: string,
              public fieldErrors: Record<string, string> = {}) { super(message); }
}
async function request<T>(path: string, method: 'POST' | 'GET' | 'PUT', body?: unknown, token?: string): Promise<T> {
  if (!baseUrl) throw new ApiError(0, 'CONFIG_ERROR', '.env의 EXPO_PUBLIC_API_BASE_URL을 설정해 주세요.');
  if (!__DEV__ && !baseUrl.startsWith('https://'))
    throw new ApiError(0, 'CONFIG_ERROR', '배포 앱은 HTTPS API 주소를 사용해야 합니다.');
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), 10000);
  try {
    const response = await fetch(baseUrl + path, {
      method, signal: controller.signal,
      headers: { ...(body !== undefined ? { 'Content-Type': 'application/json' } : {}),
                 ...(token ? { Authorization: `Bearer ${token}` } : {}) },
      ...(body !== undefined ? { body: JSON.stringify(body) } : {}),
    });
    const data = await response.json().catch(() => null);
    if (!response.ok) throw new ApiError(response.status, data?.code ?? 'HTTP_ERROR',
      data?.message ?? '요청을 처리하지 못했습니다.', data?.fieldErrors ?? {});
    if (data === null) throw new ApiError(response.status, 'INVALID_RESPONSE', '서버 응답 형식을 확인해 주세요.');
    return data as T;
  } catch (error) {
    if (error instanceof ApiError) throw error;
    if (controller.signal.aborted) throw new ApiError(0, 'TIMEOUT', '요청 시간이 초과되었습니다. 다시 시도해 주세요.');
    throw new ApiError(0, 'NETWORK_ERROR', '서버에 연결할 수 없습니다. 네트워크와 API 주소를 확인해 주세요.');
  } finally { clearTimeout(timer); }
}

  function add(raw: string) {
    if (lock.current) return;
    try { setItems(addRestriction(items, raw, type)); setCustom(''); setError(''); }
    catch (e) { setError(e instanceof Error ? e.message : '등록하지 못했습니다.'); }
  }
  async function savePreferences() {
    if (!categories.length) { setError('선호 카테고리를 1개 이상 선택해 주세요.'); return; }
    await run(async () => { const saved = await api.preferences(token, categories); setProfile(saved); setCategories(saved.preferences); setRecommendations(null); setScreen('RESTRICTIONS'); });
  }
  async function saveRestrictions(skip = false) {
    if (skip && items.length) { setError('등록한 항목이 있습니다. 삭제하거나 저장해 주세요.'); return; }
    if (!skip && custom.trim()) { setError('직접 입력한 식재료를 먼저 추가하거나 입력란을 비워 주세요.'); return; }
    await run(async () => {
      const saved = await api.restrictions(token, skip ? [] : items); setProfile(saved); setItems(saved.restrictions);
      setCustom(''); setRecommendations(null); setScreen('HOME');
    });
  }
  async function recommend() {
    if (!validCoordinates(lat, lng)) { setError('유효한 위도(-90~90)와 경도(-180~180)를 입력해 주세요.'); return; }
    await run(async () => { setRecommendations(null); const data = await api.recommendations(token, Number(lat), Number(lng)); setRecommendations(data); setScreen('RECOMMENDATIONS'); });
  }
  const button = (text: string, action: () => void, secondary = false, disabled = false) => <Pressable
    accessibilityRole="button" accessibilityLabel={text} disabled={busy || disabled} onPress={action}
    style={[s.button, secondary && s.secondary, (busy || disabled) && s.dim]}>
    <Text style={[s.buttonText, secondary && s.secondaryText]}>{text}</Text></Pressable>;
  const chip = (key: string, label: string, selected: boolean, action: () => void) => <Pressable key={key}
    accessibilityRole="checkbox" accessibilityState={{ checked: selected }} disabled={busy} onPress={action} style={[s.chip, selected && s.selectedChip]}>
    <Text style={[s.chipText, selected && s.selectedChipText]}>{label}</Text></Pressable>;
  if (initializing) return <SafeAreaView style={s.center}><ActivityIndicator color="#34704F" /><Text style={s.hint}>저장된 취향을 불러오고 있어요.</Text></SafeAreaView>;
  if (initialError) return <SafeAreaView style={s.page}><View style={s.body}><Text style={s.title}>설정을 불러오지 못했어요</Text><Text style={s.error}>{initialError}</Text>
    {button('다시 시도', () => setReload(n => n + 1))}{button('로그아웃', () => { void onLogout(); }, true)}</View></SafeAreaView>;
  return <SafeAreaView style={s.page}><StatusBar style="dark" />
    <KeyboardAvoidingView style={{ flex: 1 }} behavior={Platform.OS === 'ios' ? 'padding' : undefined}>
    <ScrollView keyboardShouldPersistTaps="handled" contentContainerStyle={s.body}>
      <View style={s.header}><Text style={s.brand}>오늘 한 끼</Text><Pressable disabled={busy} onPress={() => { void onLogout(); }} accessibilityRole="button"><Text style={s.link}>로그아웃</Text></Pressable></View>
      <Text style={s.account}>{session.user.email}</Text>
      {screen === 'PREFERENCES' && <>
        <Text style={s.step}>STEP 01 / 02 · US-02</Text><Text style={s.title}>어떤 음식을
좋아하세요?</Text>
        <Text style={s.description}>1개 이상 선택해 주세요. 선택한 카테고리를 추천 순위에 우선 반영해요.</Text>
        <View style={s.wrap}>{(Object.keys(CATEGORY_LABELS) as Category[]).map(c => chip(c, CATEGORY_LABELS[c], categories.includes(c), () => setCategories(old => old.includes(c) ? old.filter(x => x !== c) : [...old, c])))}</View>
        {button('저장하고 다음', () => { void savePreferences(); })}
        {profile?.onboardingCompleted && button('홈으로', () => { setCategories(profile.preferences); go('HOME'); }, true)}
      </>}
      {screen === 'RESTRICTIONS' && <>
        <Text style={s.step}>STEP 02 / 02 · US-03</Text><Text style={s.title}>못 먹는 재료가
있나요?</Text>
        <Text style={s.description}>알레르기와 비선호 모두 추천에서 제외해요. 목록에서 고르거나 직접 입력하세요.</Text>
        <View style={s.wrap}>{chip('ALLERGY', '알레르기', type === 'ALLERGY', () => setType('ALLERGY'))}{chip('DISLIKE', '비선호', type === 'DISLIKE', () => setType('DISLIKE'))}</View>
        <Text style={s.label}>사전 정의 목록 · 누르면 추가</Text>
        <View style={s.wrap}>{PRESET_INGREDIENTS.map(i => <Pressable key={i} disabled={busy} onPress={() => add(i)} style={s.preset} accessibilityRole="button"><Text style={s.chipText}>+ {i}</Text></Pressable>)}</View>
        <Text style={s.label}>직접 입력</Text><View style={s.inline}><TextInput value={custom} editable={!busy} onChangeText={setCustom} placeholder="예: 고수" accessibilityLabel="식재료 직접 입력" style={[s.input, { flex: 1 }]} maxLength={100} onSubmitEditing={() => add(custom)} />
          <Pressable disabled={busy} onPress={() => add(custom)} style={s.add} accessibilityRole="button"><Text style={s.buttonText}>추가</Text></Pressable></View>
        <Text style={s.label}>등록할 식재료 {items.length}개</Text>
        {items.map(item => <View key={item.ingredientName} style={s.item}><View style={{ flex: 1 }}><Text style={s.itemName}>{item.ingredientName}</Text><Text style={s.hint}>{item.type === 'ALLERGY' ? '알레르기' : '비선호'}</Text></View>
          <Pressable disabled={busy} accessibilityRole="button" accessibilityLabel={`${item.ingredientName} 삭제`} onPress={() => setItems(old => old.filter(i => i.ingredientName !== item.ingredientName))}><Text style={s.link}>삭제</Text></Pressable></View>)}
        {!items.length && <Text style={s.hint}>등록된 항목이 없어요. 건너뛰어도 괜찮아요.</Text>}
        {button('저장하고 홈으로', () => { void saveRestrictions(); })}
        {button('등록 없이 건너뛰기', () => { void saveRestrictions(true); }, true, items.length > 0 || !!custom.trim())}
        {button('선호 카테고리로 돌아가기', () => { setCustom(''); go('PREFERENCES'); }, true)}
      </>}
      {screen === 'HOME' && <>
        <Text style={s.step}>나에게 맞는 세 가지 메뉴</Text><Text style={s.title}>오늘은 무엇을
먹을까요?</Text>
        <View style={s.panel}><Text style={s.label}>저장된 선호</Text><Text style={s.summary}>{profile?.preferences.map(c => CATEGORY_LABELS[c]).join(' · ') || '아직 설정하지 않았어요'}</Text>
          <Text style={s.label}>추천에서 제외할 식재료</Text><Text style={s.summary}>{profile?.restrictions.map(i => i.ingredientName).join(' · ') || '없음'}</Text></View>
        <Text style={s.label}>시연 기준 좌표 · 실제 내 위치가 아닙니다</Text>
        <Text style={s.hint}>demo 프로필의 가상 데이터는 서울시청 인근 좌표에 있어요. US-05 GPS·외부 식당 API는 미구현입니다.</Text>
        <TextInput value={lat} editable={!busy} onChangeText={setLat} placeholder="위도" accessibilityLabel="기준 위도" keyboardType="numbers-and-punctuation" style={s.input} />
        <TextInput value={lng} editable={!busy} onChangeText={setLng} placeholder="경도" accessibilityLabel="기준 경도" keyboardType="numbers-and-punctuation" style={s.input} />
        {button('필터 적용 메뉴 보기', () => { void recommend(); })}
        {button('선호·제한 식재료 설정', () => { setCategories(profile?.preferences ?? []); setItems(profile?.restrictions ?? []); go('PREFERENCES'); }, true)}
      </>}
      {screen === 'RECOMMENDATIONS' && recommendations && <>
        <Text style={s.step}>US-07 · 반경 1 km 이내</Text><Text style={s.title}>이 메뉴는 어때요?</Text>
        <Text style={s.description}>{recommendations.message}</Text>
        <RecommendationCards key={recommendations.sessionId} items={recommendations.items} />
        <Text style={s.hint}>식당·메뉴 정보가 완전하다고 표시된 후보만 추천합니다. 실제 안전을 보증하지 않습니다.</Text>
        {button('홈으로 돌아가기', () => go('HOME'), true)}
      </>}
      {busy && <ActivityIndicator color="#34704F" style={{ marginTop: 16 }} />}
      {!!error && <Text style={s.error} accessibilityRole="alert">{error}</Text>}
    </ScrollView></KeyboardAvoidingView>
  </SafeAreaView>;
}

import React, { useRef, useState } from 'react';
import { FlatList, Pressable, StyleSheet, Text, useWindowDimensions, View } from 'react-native';
import type { MenuCard } from './types';
import { CATEGORY_LABELS } from './catalog';
export function RecommendationCards({ items }: { items: MenuCard[] }) {
  const { width } = useWindowDimensions();
  const cardWidth = Math.max(160, Math.min(width, 540) - 48);
  const list = useRef<FlatList<MenuCard>>(null);
  const [index, setIndex] = useState(0);
  const activeIndex = Math.min(index, Math.max(0, items.length - 1));
  function move(next: number) {
    if (next < 0 || next >= items.length) return;
    list.current?.scrollToIndex({ index: next, animated: true }); setIndex(next);
  }
  if (!items.length) return <Text style={s.empty}>표시할 메뉴가 없습니다. 알레르기 필터를 완화하지 않습니다.</Text>;
  return <View>
    <Text style={s.counter}>추천 {activeIndex + 1} / {items.length} · 좌우로 넘겨 보세요</Text>
    <FlatList key={String(cardWidth)} ref={list} data={items} horizontal pagingEnabled
      showsHorizontalScrollIndicator={false} keyExtractor={item => String(item.menuId)}
      style={{ width: cardWidth, flexGrow: 0 }}
      getItemLayout={(_, i) => ({ length: cardWidth, offset: cardWidth * i, index: i })}
      onMomentumScrollEnd={event => setIndex(Math.round(event.nativeEvent.contentOffset.x / cardWidth))}
      renderItem={({ item }) => <View style={{ width: cardWidth, padding: 2 }}>
        <View style={s.card} accessible accessibilityLabel={`${item.menuName}, ${item.restaurantName}, ${item.distanceMeters}미터`}>
          <View style={s.badge}><Text style={s.badgeText}>{CATEGORY_LABELS[item.category]}{item.preferred ? ' · 선호 카테고리' : ''}</Text></View>
          <Text style={s.menu}>{item.menuName}</Text><Text style={s.restaurant}>{item.restaurantName}</Text>
          <Text style={s.distance}>{item.distanceMeters.toLocaleString()} m · 직선거리</Text>
          <View style={s.divider} /><Text style={s.price}>{item.price.toLocaleString()}원</Text>
          <Text style={s.label}>등록된 식재료</Text><Text style={s.ingredients}>{item.ingredients.join(' · ')}</Text>
          <Text style={s.notice}>데이터 기준 필터 결과입니다. 실제 알레르겐·교차오염은 식당에 직접 확인하세요.</Text>
        </View>
      </View>} />
    <View style={s.nav}>
      <Pressable accessibilityRole="button" accessibilityLabel="이전 메뉴" disabled={activeIndex === 0} onPress={() => move(activeIndex - 1)} style={[s.navButton, activeIndex === 0 && s.disabled]}><Text>이전</Text></Pressable>
      <Text style={s.dots}>{items.map((_, i) => i === activeIndex ? '●' : '○').join('  ')}</Text>
      <Pressable accessibilityRole="button" accessibilityLabel="다음 메뉴" disabled={activeIndex === items.length - 1} onPress={() => move(activeIndex + 1)} style={[s.navButton, activeIndex === items.length - 1 && s.disabled]}><Text>다음</Text></Pressable>
    </View>
  </View>;
}
